\subsection{Explicit Solution of the Bipartite Semigroup and Subordinated Channel}

On $\mathbb{C}^2_S \otimes \mathbb{C}^2_M$, let $\mathcal{L}_{SM}$ be the Lindblad generator Eq.~(3) of the main text:
\begin{equation}
\label{eq:S1_L_SM}
\mathcal{L}_{SM}(\rho_{SM}) = -\ii\left[\frac{1}{2}\sigma_z \otimes \tau_z,\;\rho_{SM}\right]
+ g\Big((I_S \otimes \tau_x)\,\rho_{SM}\,(I_S \otimes \tau_x) - \rho_{SM}\Big), \qquad g > 0.
\end{equation}
Because $(I_S \otimes \tau_x)^\dagger = I_S \otimes \tau_x$ and $(I_S \otimes \tau_x)^2 = I_S \otimes I_M$, $\mathcal{L}_{SM}$ is in exact Gorini--Kossakowski--Sudarshan--Lindblad (GKSL) form with a single Hermitian jump operator $\sqrt{2g}(I_S \otimes \tau_x)$, guaranteeing that $\Lambda_{SM}(\tau) := \ee^{\tau\mathcal{L}_{SM}}$ is a trace-preserving, unital, completely positive (CPTP) contraction semigroup for all $\tau \ge 0$.

Decompose $\rho_{SM}(\tau) = \sum_{a,b\in\{0,1\}} |a\rangle\langle b|_S \otimes R_{ab}(\tau)$ in the $\sigma_z$-eigenbasis ($\sigma_z|0\rangle = +|0\rangle$, $\sigma_z|1\rangle = -|1\rangle$). Since $\frac{1}{2}\sigma_z|a\rangle = \frac{(-1)^a}{2}|a\rangle$, Eq.~\eqref{eq:S1_L_SM} decouples into four $2\times 2$ blocks on $\mathcal{H}_M$:
\begin{align}
\dot{R}_{00} &= -\frac{\ii}{2}[\tau_z, R_{00}] + g(\tau_x R_{00}\tau_x - R_{00}), \qquad
\dot{R}_{11} = +\frac{\ii}{2}[\tau_z, R_{11}] + g(\tau_x R_{11}\tau_x - R_{11}), \label{eq:S1_pop}\\
\dot{R}_{01} &= -\frac{\ii}{2}\{\tau_z, R_{01}\} + g(\tau_x R_{01}\tau_x - R_{01}), \qquad
R_{10}(\tau) = R_{01}(\tau)^\dagger. \label{eq:S1_coh}
\end{align}
Taking the trace over $M$ in Eq.~\eqref{eq:S1_pop} gives $\frac{\dd}{\dd\tau}\Tr_M(R_{aa}(\tau)) = 0$, confirming exact population preservation $\langle a|\Phi_{1,g}(\tau)\rho_S|a\rangle = \langle a|\rho_S|a\rangle$. Initializing $\rho_{SM}(0) = \rho_S \otimes \frac{1}{2}I_M$ sets $R_{01}(0) = \frac{1}{2}\rho_{01} I_M$, which is diagonal in the $\tau_z$-eigenbasis $\{|m_+\rangle, |m_-\rangle\}$ ($\tau_z|m_\pm\rangle = \pm|m_\pm\rangle$, $\tau_x|m_\pm\rangle = |m_\mp\rangle$). Because both the anti-commutator $-\frac{\ii}{2}\{\tau_z, \cdot\}$ and the bit-flip $\tau_x \cdot \tau_x$ preserve the subspace of $\tau_z$-diagonal matrices, $R_{01}(\tau) = v_+(\tau)|m_+\rangle\langle m_+| + v_-(\tau)|m_-\rangle\langle m_-|$ remains strictly diagonal for all $\tau \ge 0$, with components obeying
\begin{equation}
\label{eq:S1_v_ode}
\frac{\dd}{\dd\tau}\begin{pmatrix} v_+ \\ v_- \end{pmatrix}
= \begin{pmatrix} -g - \ii & g \\ g & -g + \ii \end{pmatrix}\begin{pmatrix} v_+ \\ v_- \end{pmatrix}, \qquad
\begin{pmatrix} v_+(0) \\ v_-(0) \end{pmatrix} = \frac{\rho_{01}}{2}\begin{pmatrix} 1 \\ 1 \end{pmatrix}.
\end{equation}
Taking the Laplace transform $\tilde{v}_\pm(u) = \int_0^\infty \ee^{-u\tau} v_\pm(\tau)\,\dd\tau$ ($\operatorname{Re} u > 0$) gives
\[
\begin{pmatrix} u + g + \ii & -g \\ -g & u + g - \ii \end{pmatrix}\begin{pmatrix} \tilde{v}_+(u) \\ \tilde{v}_-(u) \end{pmatrix}
= \frac{\rho_{01}}{2}\begin{pmatrix} 1 \\ 1 \end{pmatrix}.
\]
The matrix determinant equals $(u+g)^2 + 1 - g^2 = u^2 + 2gu + 1$. Inverting the $2\times 2$ matrix yields
\begin{equation}
\label{eq:S1_v_lap}
\tilde{v}_\pm(u) = \frac{\rho_{01}}{2}\,\frac{u + 2g \mp \ii}{u^2 + 2gu + 1}
\quad\Longrightarrow\quad
\Tr_M\big[\tilde{R}_{01}(u)\big] = \tilde{v}_+(u) + \tilde{v}_-(u) = \frac{u + 2g}{u^2 + 2gu + 1}\,\rho_{01} = \tilde{G}_{1,g}(u)\,\rho_{01}.
\end{equation}
Subordinating the operational clock $\tau$ by the inverse $\mu$-stable hitting-time density $p_\mu(\tau,t) \ge 0$ ($\int_0^\infty p_\mu(\tau,t)\,\dd\tau = 1$, $\int_0^\infty \ee^{-st} p_\mu(\tau,t)\,\dd t = s^{\mu-1}\ee^{-\tau s^\mu}$ for $\operatorname{Re} s > 0$, $\mu \in (0,1]$) produces the reduced channel Eq.~(5) of the main text with
\begin{equation}
\label{eq:S1_G_sub}
G_{\mu,g}(t) = \int_0^\infty p_\mu(\tau,t)\,G_{1,g}(\tau)\,\dd\tau, \qquad
\tilde{G}_{\mu,g}(s) = \int_0^\infty s^{\mu-1}\ee^{-\tau s^\mu} G_{1,g}(\tau)\,\dd\tau
= s^{\mu-1}\tilde{G}_{1,g}(s^\mu) = \frac{s^{\mu-1}(s^\mu + 2g)}{s^{2\mu} + 2g s^\mu + 1}.
\end{equation}

\subsection{Choi Matrix Spectrum and Global BLP Optimality (Proposition 1)}

For the unital pure-dephasing map $\Phi_{\mu,g}(t)(\rho) = \frac{1+G_{\mu,g}(t)}{2}\rho + \frac{1-G_{\mu,g}(t)}{2}\sigma_z\rho\sigma_z$, the normalized $4\times 4$ Choi--Jamio\l kowski state $J(\Phi_{\mu,g}(t)) := (\Phi_{\mu,g}(t)\otimes \mathrm{id}_2)(|\Phi^+\rangle\langle\Phi^+|)$ with $|\Phi^+\rangle = \frac{1}{\sqrt{2}}(|00\rangle + |11\rangle)$ reads explicitly in the computational basis $\{|00\rangle, |01\rangle, |10\rangle, |11\rangle\}$:
\begin{equation}
\label{eq:S1_choi}
J(\Phi_{\mu,g}(t)) = \frac{1}{2}\begin{pmatrix} 1 & 0 & 0 & G_{\mu,g}(t) \\ 0 & 0 & 0 & 0 \\ 0 & 0 & 0 & 0 \\ G_{\mu,g}(t) & 0 & 0 & 1 \end{pmatrix},
\end{equation}
whose eigenvalues are $\big\{\frac{1+G_{\mu,g}(t)}{2}, \frac{1-G_{\mu,g}(t)}{2}, 0, 0\big\}$. Since $\Phi_{\mu,g}(t)$ is a probability-weighted mixture of CPTP maps $\Tr_M \circ \ee^{\tau\mathcal{L}_{SM}}(\cdot\otimes\rho_M^{\mathrm{eq}})$, $J(\Phi_{\mu,g}(t)) \ge 0$ holds identically, proving $|G_{\mu,g}(t)| \le 1$ for all $(\mu,g) \in (0,1]\times (0,\infty)$ and all $t \ge 0$.

For any pair of qubit states $\rho_j = \frac{1}{2}(I + \mathbf{r}_j\cdot\boldsymbol{\sigma})$ ($j=1,2$, $\|\mathbf{r}_j\|_2 \le 1$) with Bloch difference $\Delta\mathbf{r} = (\Delta x, \Delta y, \Delta z)$ and $r_\perp^2 := \Delta x^2 + \Delta y^2 \le \|\Delta\mathbf{r}\|_2^2 \le 4$, the evolved difference has Bloch vector $(G_{\mu,g}(t)\Delta x, G_{\mu,g}(t)\Delta y, \Delta z)$, so $D_{\mathrm{tr}}(\Phi_t\rho_1, \Phi_t\rho_2) = \frac{1}{2}\sqrt{G(t)^2 r_\perp^2 + \Delta z^2}$. Differentiating wherever $G(t)^2 r_\perp^2 + \Delta z^2 > 0$ gives Eq.~(7) of the main text:
\begin{equation}
\label{eq:S1_sigma_blp}
\sigma_{\mathrm{BLP}}(t; \rho_1, \rho_2) = \frac{G(t)\dot{G}(t)\,r_\perp^2}{2\sqrt{G(t)^2 r_\perp^2 + \Delta z^2}}.
\end{equation}
If $G(t)\dot{G}(t) \le 0$, every pair has $\sigma_{\mathrm{BLP}}(t) \le 0$. If $G(t)\dot{G}(t) > 0$, then $G(t) \neq 0$, and dropping $\Delta z^2 \ge 0$ plus using $r_\perp \le 2$ gives
\[
\sigma_{\mathrm{BLP}}(t; \rho_1, \rho_2)
\le \frac{G(t)\dot{G}(t)\,r_\perp^2}{2|G(t)| r_\perp}
= \frac{r_\perp}{2}\,\sgn(G(t))\,\dot{G}(t)
\le \sgn(G(t))\,\dot{G}(t) = \frac{\dd}{\dd t}|G(t)|,
\]
attained simultaneously at every $t \ge 0$ by the antipodal equatorial pair $|\pm\rangle = \frac{1}{\sqrt{2}}(|0\rangle\pm |1\rangle)$ ($\Delta x = 2, \Delta y = \Delta z = 0$). Integrating the positive part over $[0,\infty)$ proves $\NBLP(\mu,g) = \int_{\frac{\dd}{\dd t}|G_{\mu,g}(t)| > 0} \frac{\dd}{\dd t}|G_{\mu,g}(t)|\,\dd t$.

\subsection{Closed-Form Mittag-Leffler Decomposition Across \texorpdfstring{$g < 1$, $g = 1$, and $g > 1$}{g < 1, g = 1, and g > 1}}

Setting $u = s^\mu$, partial-fraction decomposition of $\frac{u+2g}{u^2+2gu+1}$ separates into three regimes:
\begin{enumerate}[label=(\roman*),leftmargin=1.8em,itemsep=2pt]
\item \textbf{Underdamped ($0 < g < 1$, $\omega = \sqrt{1-g^2}$, $u_\pm = -g \pm \ii\omega$):}
\[
\frac{u+2g}{(u-u_+)(u-u_-)} = \frac{c_+}{u-u_+} + \frac{c_-}{u-u_-}, \qquad
c_+ = \frac{u_+ + 2g}{u_+ - u_-} = \frac{g + \ii\omega}{2\ii\omega} = \frac{1}{2}\left(1 - \ii\frac{g}{\omega}\right), \qquad c_- = \overline{c_+}.
\]
Applying $\mathcal{L}^{-1}\!\left[\frac{s^{\mu-1}}{s^\mu - a}\right](t) = E_\mu(a t^\mu)$ and $\frac{\dd}{\dd t} E_\mu(a t^\mu) = a t^{\mu-1} E_{\mu,\mu}(a t^\mu)$, together with $c_+ u_+ = \frac{1}{2}(1-\ii g/\omega)(-g+\ii\omega) = \frac{\ii}{2\omega}$, yields
\begin{equation}
\label{eq:S2_ml_under}
G_{\mu,g}(t) = \operatorname{Re}\!\left[\left(1 - \ii\frac{g}{\omega}\right)E_\mu(u_+ t^\mu)\right], \qquad
\dot{G}_{\mu,g}(t) = 2\operatorname{Re}\!\left[\frac{\ii}{2\omega} t^{\mu-1} E_{\mu,\mu}(u_+ t^\mu)\right] = -\frac{t^{\mu-1}}{\omega}\operatorname{Im}\!\left[E_{\mu,\mu}(u_+ t^\mu)\right].
\end{equation}
\item \textbf{Exceptional point ($g = 1$):} $\frac{u+2}{(u+1)^2} = \frac{1}{u+1} + \frac{1}{(u+1)^2}$. Using $\mathcal{L}^{-1}\!\left[\frac{s^{\mu-1}}{(s^\mu+1)^2}\right](t) = \sum_{k=0}^\infty \frac{(k+1)(-1)^k t^{\mu(k+1)}}{\Gamma(\mu k + \mu + 1)} = \frac{t^\mu}{\mu} E_{\mu,\mu}(-t^\mu)$ (since $\frac{k+1}{\Gamma(\mu(k+1)+1)} = \frac{1}{\mu\Gamma(\mu(k+1))}$) gives
\begin{equation}
\label{eq:S2_ml_crit}
G_{\mu,1}(t) = E_\mu(-t^\mu) + \frac{t^\mu}{\mu} E_{\mu,\mu}(-t^\mu), \qquad
\dot{G}_{\mu,1}(t) = -t^{2\mu-1}\sum_{k=0}^\infty \frac{(k+1)(-t^\mu)^k}{\Gamma(\mu(k+2))}.
\end{equation}
\item \textbf{Overdamped ($g > 1$, $\Omega = \sqrt{g^2-1}$, $\omega_\pm = g \pm \Omega > 0$, $\omega_+\omega_- = 1$):} $\frac{u+2g}{(u+\omega_-)(u+\omega_+)} = \frac{1}{2\Omega}\!\left(\frac{\omega_+}{u+\omega_-} - \frac{\omega_-}{u+\omega_+}\right)$, giving
\begin{equation}
\label{eq:S2_ml_over}
G_{\mu,g}(t) = \frac{\omega_+ E_\mu(-\omega_- t^\mu) - \omega_- E_\mu(-\omega_+ t^\mu)}{2\Omega}, \qquad
\dot{G}_{\mu,g}(t) = -\frac{t^{\mu-1}}{2\Omega}\big[E_{\mu,\mu}(-\omega_- t^\mu) - E_{\mu,\mu}(-\omega_+ t^\mu)\big],
\end{equation}
since $\omega_+(-\omega_-) = -1$ and $-\omega_-(-\omega_+) = +1$.
\end{enumerate}

\subsection{Bromwich--Hankel Contour Deformation and Exact Spectral Density \texorpdfstring{$K_{\mu,g}(r)$}{K\_\{mu,g\}(r)}}

For $\mu \in (0,1)$, choose the principal branch of $s^\mu = |s|^\mu \ee^{\ii\mu\arg s}$ with $\arg s \in (-\pi, \pi)$ and branch cut along $(-\infty, 0]$. On the large circle $|s| = R \to \infty$ in the left half-plane, $|\tilde{G}_{\mu,g}(s)| = O(R^{-1})$ uniformly, so Jordan's lemma annihilates the outer circular arcs for every $t > 0$. On the small circle $|s| = \varepsilon \to 0^+$, $|\tilde{G}_{\mu,g}(s)| = O(\varepsilon^{\mu-1})$, so $\big|\int_{|s|=\varepsilon} \ee^{st}\tilde{G}_{\mu,g}(s)\,\dd s\big| = O(\varepsilon^\mu) \to 0$. Deforming the Bromwich line $\operatorname{Re} s = c > 0$ onto the Hankel keyhole contour hugging the upper bank $s = r\ee^{+\ii\pi}$ ($r:\infty\to 0$) and lower bank $s = r\ee^{-\ii\pi}$ ($r:0\to\infty$) yields by Cauchy's residue theorem:
\begin{equation}
\label{eq:S2_keyhole}
G_{\mu,g}(t) = \sum_{\substack{s_* \text{ pole}\\|\arg s_*| < \pi}} \operatorname{Res}_{s=s_*}\!\left[\tilde{G}_{\mu,g}(s)\ee^{st}\right]
+ \frac{1}{2\pi\ii}\int_0^\infty \ee^{-rt}\Big(\tilde{G}_{\mu,g}(r\ee^{-\ii\pi}) - \tilde{G}_{\mu,g}(r\ee^{+\ii\pi})\Big)\dd r.
\end{equation}
Because $\tilde{G}_{\mu,g}(\bar{s}) = \overline{\tilde{G}_{\mu,g}(s)}$ by Schwarz reflection, $\tilde{G}_{\mu,g}(r\ee^{-\ii\pi}) - \tilde{G}_{\mu,g}(r\ee^{+\ii\pi}) = 2\ii\operatorname{Im}\tilde{G}_{\mu,g}(r\ee^{-\ii\pi})$, giving $G_{\mathrm{cut}}(t) = \int_0^\infty K_{\mu,g}(r)\ee^{-rt}\,\dd r$ with $K_{\mu,g}(r) = \frac{1}{\pi}\operatorname{Im}\tilde{G}_{\mu,g}(r\ee^{-\ii\pi})$. Writing $z = u\ee^{-\ii\pi\mu}$ ($u = r^\mu > 0$), expanding $-z(z+2g)(\bar{z}^2+2g\bar{z}+1)$ over $\mathbb{C}$, and noting that $|z|^4 + 4g^2|z|^2 + 2g|z|^2(z+\bar{z}) \in \mathbb{R}$ leaves $-\operatorname{Im}(z^2+2gz) = u^2\sin(2\pi\mu) + 2gu\sin(\pi\mu) = 2u\sin(\pi\mu)(u\cos(\pi\mu)+g)$ exactly (verified symbolically in Lean~4, \leanid{hankel\_propagator\_numerator\_im}), giving Eqs.~(13)--(15) of the main text.

\textbf{Factorization of $D_{\mu,g}(r)$ and Boundary Cancellation at $\mu = \mupole(g)$:}
For $g \in (0,1)$ with $\theta = \arccos g \in (0, \pi/2)$ and $\mupole(g) = 1 - \theta/\pi \in (1/2, 1)$, we have $u_\pm = \ee^{\pm\ii(\pi-\theta)}$, so
\[
|z - u_+|^2 = |u\ee^{-\ii\pi\mu} - \ee^{\ii(\pi-\theta)}|^2 = u^2 + 1 - 2u\cos(\pi\mu + \pi - \theta) = (u-1)^2 + 4u\cos^2\!\left(\frac{\pi\mu - \theta}{2}\right),
\]
\[
|z - u_-|^2 = |u\ee^{-\ii\pi\mu} - \ee^{-\ii(\pi-\theta)}|^2 = u^2 + 1 - 2u\cos(\pi\mu - \pi + \theta) = (u-1)^2 + 4u\sin^2\!\left(\frac{\pi(\mu - \mupole(g))}{2}\right).
\]
Multiplying establishes Eq.~(16) of the main text. Hence $D_{\mu,g}(r) \ge 0$ everywhere, vanishing on $(0,\infty)$ if and only if $u = 1$ ($r=1$) and $\mu = \mupole(g)$. Right on $\mu = \mupole(g)$, $\cos(\pi\mupole) = -g$, so $u\cos(\pi\mupole) + g = -g(u-1)$ cancels one power of $(u-1)$ against $(u-1)^2$, leaving a simple pole at $r = 1$:
\begin{equation}
\label{eq:S2_K_pole_boundary}
K_{\mupole(g),g}(r) = -\frac{2g\omega}{\pi}\,\frac{r^{\mupole-1}}{(r^{\mupole}-1)\big(r^{2\mupole} + (2-4g^2)r^{\mupole} + 1\big)},
\end{equation}
with residue $\operatorname{Res}_{r=1} K_{\mupole,g}(r) = -\frac{2g\omega}{\pi\cdot\mupole\cdot 4(1-g^2)} = -\frac{g}{2\pi\mupole\omega}$. Indenting the keyhole contour symmetrically around $s = -1$ (Sokhotski--Plemelj half-residue $\frac{1}{2}\lim_{\mu\downarrow\mupole}G_{\mathrm{pole}}(t) = \frac{1}{2\mupole}\ee^{-t}$) gives $G_{\mupole,g}(t) = \mathrm{PV}_r\!\int_0^\infty K_{\mupole,g}(r)\ee^{-rt}\,\dd r + \frac{1}{2\mupole}\ee^{-t}$.

\subsection{Complete Dominated Convergence Step on \texorpdfstring{$\mu = \mupole(g)$}{mu = mu\_pole(g)} (Theorem 4)}

In the open interior of $\mathcal{W}_{\mathrm{PF}}$ with $\mu \in (1/2, 1)$, $\phi_{\mu,g}(r) := r K_{\mu,g}(r)$ satisfies $|\phi_{\mu,g}(r)| = O(r^\mu)$ as $r \to 0^+$ and $O(r^{-2\mu})$ as $r \to \infty$, so $\phi_{\mu,g} \in C((0,\infty)) \cap L^1((0,\infty))$. Differentiating $G_{\mu,g}(t) = \int_0^\infty K_{\mu,g}(r)\ee^{-rt}\,\dd r$ under the integral sign yields $-\dot{G}_{\mu,g}(t) = \int_0^\infty \phi_{\mu,g}(r)\ee^{-rt}\,\dd r$, and taking $t \to 0^+$ gives $\int_0^\infty \phi_{\mu,g}(r)\,\dd r = -\dot{G}_{\mu,g}(0^+) = 0$ (because $1 - G_{\mu,g}(t) \sim t^{2\mu}/\Gamma(2\mu+1)$ and $2\mu > 1$). Since $\phi_{\mu,g}(r)$ changes sign strictly once at $r_0(\mu) = (g/|\cos(\pi\mu)|)^{1/\mu}$ from positive to negative and is not zero almost everywhere, Lemma~3 of the main text establishes $-\dot{G}_{\mu,g}(t) = \int_0^\infty \phi_{\mu,g}(r)(\ee^{-rt}-\ee^{-r_0(\mu)t})\,\dd r > 0$.

Right on $\mu = \mupole(g)$, the singular factor $(r^{\mupole}-1)$ in Eq.~\eqref{eq:S2_K_pole_boundary} is cancelled identically by $(\ee^{-rt} - \ee^{-t}) = -t\ee^{-t}(r-1) + O((r-1)^2)$:
\begin{equation}
\label{eq:S2_removable_limit}
\lim_{r\to 1} \phi_{\mupole(g),g}(r)(\ee^{-rt} - \ee^{-t})
= \left(-\frac{2g\omega}{\pi\cdot 4\omega^2}\right)\left(\frac{-t\ee^{-t}}{\mupole(g)}\right)
= \frac{g\,t\,\ee^{-t}}{2\pi\mupole(g)\omega} > 0.
\end{equation}
Together with the uniform $L^1$ envelope $M_t \mathbf{1}_{[1/2,3/2]} + C_3(r^{\mu_1}\mathbf{1}_{(0,1/2)} + r^{-2\mu_1}\mathbf{1}_{(3/2,\infty)})$ derived in Theorem~4 of the main text, Lebesgue's Dominated Convergence Theorem yields $-\dot{G}_{\mupole(g),g}(t) = \int_0^\infty \phi_{\mupole(g),g}(r)(\ee^{-rt}-\ee^{-t})\,\dd r > 0$ strictly for every $t > 0$.

\subsection{Two-Term Watson Expansion of the Super-Exponential Buffer Closing Law}

Set $\mu = 1 - \epsilon$ with $\epsilon \to 0^+$ at fixed $g \in (0,1)$. Then $\sin(\pi\mu) = \sin(\pi\epsilon) = \pi\epsilon + O(\epsilon^3)$, $\cos(\pi\mu) = -\cos(\pi\epsilon) = -1 + O(\epsilon^2)$, and $r^\mu = r(1 - \epsilon\ln r + O(\epsilon^2))$, so Eqs.~(13)--(14) of the main text expand uniformly on compact subsets of $(0,1)$ to
\begin{equation}
\label{eq:S3_K_eps}
K_{1-\epsilon,g}(r) = 2\epsilon\,\frac{g - r}{(r^2 - 2gr + 1)^2} + O(\epsilon^2), \qquad
\phi_{1-\epsilon,g}(r) = 2\epsilon\,\frac{r(g - r)}{(r^2 - 2gr + 1)^2} + O(\epsilon^2).
\end{equation}
Expanding the rational factor about $r = 0$:
\[
\frac{r(g-r)}{(1 - 2gr + r^2)^2}
= r(g-r)\big(1 + 4gr + O(r^2)\big)
= g r + (4g^2 - 1)r^2 + O(r^3).
\]
By Watson's lemma for large $t_1$, integrating $-\dot{G}_{\mathrm{cut}}(t_1) = \int_0^\infty \phi_{1-\epsilon,g}(r)\ee^{-r t_1}\,\dd r$ term-by-term via $\int_0^\infty r^n \ee^{-r t_1}\,\dd r = n!/t_1^{n+1}$ gives
\begin{equation}
\label{eq:S3_watson2}
-\dot{G}_{\mathrm{cut}}(t_1) = 2\epsilon\left[\frac{g}{t_1^2} + \frac{2(4g^2 - 1)}{t_1^3} + O(t_1^{-4})\right] + O(\epsilon^2).
\end{equation}
Meanwhile, at $\mu = 1$ the pole contribution equals $\dot{G}_{1,g}(t) = -\frac{1}{\omega}\ee^{-gt}\sin(\omega t)$, whose first positive maximum occurs where $\ddot{G}_{1,g}(t) = 0 \iff \tan(\omega t) = \omega/g = \tan\theta \implies \omega t_1 = \pi + \theta$, i.e. $t_1(g) = (\pi + \arccos g)/\sqrt{1-g^2}$. Evaluating $\dot{G}_{1,g}(t_1)$ gives
\[
\dot{G}_{1,g}(t_1) = -\frac{1}{\omega}\ee^{-g t_1}\sin(\pi+\theta) = +\frac{\sin\theta}{\omega}\ee^{-g t_1} = \ee^{-g t_1}.
\]
Setting $\dot{G}_{\mathrm{pole}}(t_1) + \dot{G}_{\mathrm{cut}}(t_1) = 0$ yields Eq.~(21) of the main text:
\begin{equation}
\label{eq:S3_asymp_full}
1 - \mucrit(g) \sim \frac{t_1(g)^2}{2g + 4(4g^2-1)/t_1(g)}\,\ee^{-g t_1(g)}.
\end{equation}
Writing $g = 1 - \delta$ ($\delta \to 0^+$), we have $\omega = \sqrt{2\delta}(1 + O(\delta))$ and $\arccos(1-\delta) = \sqrt{2\delta}(1 + O(\delta))$, so $\frac{g\arccos g}{\sqrt{1-g^2}} = 1 + O(\delta)$ and $t_1(g) = \frac{\pi}{\sqrt{1-g^2}}(1 + O(\sqrt{\delta}))$, giving the leading closed form $\frac{\pi^2}{2\ee(1-g^2)}\exp(-\pi g/\sqrt{1-g^2})$.

\subsection{Finiteness of Backflow Lobes and Non-Invertibility Zero Crossings for \texorpdfstring{$\mu < 1$}{mu < 1}}

\begin{theorem}[Global Finiteness for Every $\mu \in (0,1)$]
\label{thm:S3_finite}
For every $g \in (0,1)$ and every $\mu \in (0,1)$, both the backflow set $\{t > 0 : \sigma_{\mathrm{BLP}}(t) > 0\}$ and the non-invertibility zero set $\{t > 0 : G_{\mu,g}(t) = 0\}$ are bounded subsets of $(0,\infty)$, consisting of at most finitely many intervals and finitely many isolated zeros, respectively.
\end{theorem}
\begin{proof}
If $\mu \le \mupole(g)$, both sets are empty by Theorem~4 of the main text. If $\mu \in (\mupole(g), 1)$, then $\psi = (\pi-\theta)/\mu \in (\pi/2, \pi)$ implies $\alpha = -\cos\psi > 0$, so by Eq.~(19) of the main text $|G_{\mathrm{pole}}(t)| \le \frac{1}{\mu\sin\theta}\ee^{-\alpha t}$ and $|\dot{G}_{\mathrm{pole}}(t)| \le \frac{1}{\mu\sin\theta}\ee^{-\alpha t}$ (note also that $\dot{G}_{\mathrm{pole}}(0) = \frac{\sin(\psi+\theta)}{\mu\sin\theta} = \frac{\cos\psi + (g/\omega)\sin\psi}{\mu} < 0$ on $(\mupole(g), 1)$ and vanishes at $\mu = 1$). Conversely, from $K_{\mu,g}(r) = \frac{2g\sin(\pi\mu)}{\pi} r^{\mu-1}(1 + O(r^\mu))$ as $r \to 0^+$, Watson's lemma yields $G_{\mathrm{cut}}(t) \sim \frac{2g}{\Gamma(1-\mu)} t^{-\mu} > 0$ and $\dot{G}_{\mathrm{cut}}(t) \sim -\frac{2g\mu}{\Gamma(1-\mu)} t^{-(\mu+1)} < 0$ as $t \to \infty$. Because $\ee^{-\alpha t} = o(t^{-(\mu+1)})$ for any $\alpha > 0$, there exists $T_*(\mu,g) < \infty$ such that for all $t \ge T_*$, both $G_{\mu,g}(t) > 0$ and $\dot{G}_{\mu,g}(t) < 0$ strictly (so $\sigma_{\mathrm{BLP}}(t) < 0$). Because $G_{\mu,g}$ and $\dot{G}_{\mu,g}$ are real-analytic on $(0,\infty)$ (as sums of entire functions of $t^\mu$ via Eq.~\eqref{eq:S2_ml_under}), they can have at most finitely many zeros on the compact interval $[t_0, T_*]$ (with $t_0 > 0$ chosen inside the initial monotonic window $G_{\mu,g} \sim 1 - t^{2\mu}/\Gamma(2\mu+1)$), proving finiteness of both sets.
\end{proof}

\subsection{Exact Analytical Derivative \texorpdfstring{$\kappa_\mu'(x)$}{kappa\_mu'(x)}, Global Subadditivity, and 2D Criticality System}

Recall $\kappa_\mu(x) := \frac{x\,E_{\mu,\mu}(-x)}{E_\mu(-x)}$ for $x > 0$ and $\mu \in (0,1)$, where $f(x) := E_\mu(-x)$ and $g(x) := x E_{\mu,\mu}(-x)$ satisfy $f'(x) = -\frac{1}{\mu} E_{\mu,\mu}(-x) = -\frac{g(x)}{\mu x}$ (directly from differentiating $E_\mu(-x) = \sum_{k=0}^\infty \frac{(-x)^k}{\Gamma(\mu k+1)}$ termwise: $f'(x) = -\sum_{m=0}^\infty \frac{(m+1)(-x)^m}{\Gamma(\mu m + \mu + 1)} = -\frac{1}{\mu}\sum_{m=0}^\infty \frac{(-x)^m}{\Gamma(\mu m+\mu)}$ since $\frac{m+1}{\Gamma(\mu(m+1)+1)} = \frac{1}{\mu\Gamma(\mu(m+1))}$). Differentiating $\kappa_\mu(x) = g(x)/f(x)$ therefore yields the exact analytical identity
\begin{equation}
\label{eq:S4_kappa_prime}
\kappa_\mu'(x)
= \frac{g'(x)f(x) - g(x)f'(x)}{f(x)^2}
= \frac{g'(x)}{f(x)} + \frac{\kappa_\mu(x)^2}{\mu x},
\end{equation}
where $g'(x) = \sum_{k=0}^\infty \frac{(k+1)(-x)^k}{\Gamma(\mu k + \mu)}$ is evaluated via Hankel contour quadrature (for large $x$) or convergent series (for moderate $x$) at 75- and 95-digit working precision in \texttt{pauli\_hierarchy.py}.

\begin{lemma}[Analytic Global Subadditivity of $\kappa_\mu(x)$ via Pollard Log-Convexity]
\label{lem:S4_pollard_subadd}
For every $\mu \in (0,1)$, the secant function $x \mapsto \kappa_\mu(x)/x$ is strictly decreasing on $(0,\infty)$, and consequently $\kappa_\mu$ is strictly subadditive globally on $(0,\infty)$: $\kappa_\mu(x + y) < \kappa_\mu(x) + \kappa_\mu(y)$ for all $x, y > 0$.
\end{lemma}
\begin{proof}
By Pollard's theorem, $E_\mu(-x) = \int_0^\infty \ee^{-rx}\,\dd\nu_\mu(r)$ is the Laplace transform of a positive non-degenerate probability measure $\nu_\mu$ on $(0,\infty)$ for every $\mu \in (0,1)$. By the strict Cauchy--Schwarz inequality,
\[
E_\mu(-x)\,E_\mu''(-x) - \big(E_\mu'(-x)\big)^2
= \left(\int_0^\infty \ee^{-rx}\,\dd\nu_\mu\right)\!\left(\int_0^\infty r^2 \ee^{-rx}\,\dd\nu_\mu\right)
- \left(\int_0^\infty r\,\ee^{-rx}\,\dd\nu_\mu\right)^2 > 0,
\]
so $\frac{\dd^2}{\dd x^2}\ln E_\mu(-x) > 0$ strictly for all $x > 0$. Because $\frac{\kappa_\mu(x)}{x} = \frac{E_{\mu,\mu}(-x)}{E_\mu(-x)} = -\mu\frac{\dd}{\dd x}\ln E_\mu(-x)$, we obtain
\begin{equation}
\label{eq:S4_secant_neg}
\frac{\dd}{\dd x}\!\left(\frac{\kappa_\mu(x)}{x}\right)
= -\mu\,\frac{\dd^2}{\dd x^2}\ln E_\mu(-x) < 0 \qquad \forall x > 0.
\end{equation}
For any $x, y > 0$, strict decreasingness of $\kappa_\mu(r)/r$ gives $\frac{\kappa_\mu(x+y)}{x+y} < \frac{\kappa_\mu(x)}{x}$ and $\frac{\kappa_\mu(x+y)}{x+y} < \frac{\kappa_\mu(y)}{y}$; multiplying by $x$ and $y$ respectively and adding yields $\kappa_\mu(x+y) < \kappa_\mu(x) + \kappa_\mu(y)$.
\end{proof}

Solving the 2D criticality system $\big(\kappa_\mu(x) - 2\mu,\; \kappa_\mu'(x)\big) = (0, 0)$ via damped Newton iteration yields the following verified constants (75-digit vs.\ 95-digit agreement $|\Delta\mustar| = 3.00\times 10^{-77}$, $|\Delta x_*| = 5.27\times 10^{-77}$, residuals $|\kappa_{\mustar}(x_*)-2\mustar| = 1.82\times 10^{-96}$ and $|\kappa'_{\mustar}(x_*)| = 6.44\times 10^{-98}$):

\subsection{Complete Asymmetric Reduction Across \texorpdfstring{$\mu \in (1/2, \mustar]$}{mu in (1/2, mu\_*]} (Proposition 7)}

Let $0 \le u = \lambda_y t^\mu \le v = \lambda_z t^\mu$ and $w = \lambda_x t^\mu \in [v-u, v+u]$ be any triangle-valid triple at $\mu \in (1/2, \mustar]$ (under the unimodal profile of $\kappa_\mu$ on $(1/2, \mustar]$ with unique crossing $x_0(\mu)$ of $\mu$ and unique peak $x_{\mathrm{peak}}(\mu) > x_0(\mu)$, consistent with the large-$x$ expansion $\kappa_\mu(x) = \mu[1 - \frac{\Gamma(1-\mu)}{\Gamma(1-2\mu)}x^{-1} + O(x^{-2})] > \mu$). Every case of $\kappa_\mu(u) + \kappa_\mu(v) - \kappa_\mu(w) \ge 0$ is exhausted cleanly:
\begin{enumerate}[label=(\roman*),leftmargin=1.8em,itemsep=1pt]
\item \textbf{$u \ge x_0(\mu)$ (hence $v \ge x_0(\mu)$):} $\kappa_\mu(u) \ge \mu$ and $\kappa_\mu(v) \ge \mu$, giving $\kappa_\mu(u) + \kappa_\mu(v) - \kappa_\mu(w) \ge 2\mu - \max_{x>0}\kappa_\mu(x) \ge 0$ analytically for all $\mu \le \mustar$.
\item \textbf{$u < x_0(\mu)$ and $w \ge v$:} Write $w = v + \delta$ with $0 \le \delta \le u < x_0(\mu) < x_{\mathrm{peak}}(\mu)$. (If in addition $v \ge x_{\mathrm{peak}}(\mu)$, decreasingness on $[x_{\mathrm{peak}}(\mu), \infty)$ already yields $\kappa_\mu(w) \le \kappa_\mu(v)$ directly.)
\item \textbf{$u < x_0(\mu)$, $v \le x_{\mathrm{peak}}(\mu)$, and $v - u \le w < v$:} Increasing monotonicity on $[0, x_{\mathrm{peak}}(\mu)]$ gives $\kappa_\mu(w) \le \kappa_\mu(v) < \kappa_\mu(v) + \kappa_\mu(u)$.
\item \textbf{$u < x_0(\mu)$, $v > x_{\mathrm{peak}}(\mu)$, and $v - u \le w < v$:} Setting $\tilde{w} := \max(w, x_{\mathrm{peak}}(\mu))$ gives $\kappa_\mu(w) \le \kappa_\mu(\tilde{w})$ and $v - \tilde{w} \le v - w \le u$, while Lemma~\ref{lem:S4_pollard_subadd} guarantees $\kappa_\mu(u) \ge (\mu/x_0(\mu))u$ analytically on $(0, x_0(\mu)]$. \[

\le \kappa_\mu(\tilde{w}) - \kappa_\mu(v)
\le L_{\mathrm{down}}(\mu)\,u
< \frac{\mu}{x_0(\mu)}\,u

\end{enumerate}

\subsection{Proof of the Nested Inclusions \texorpdfstring{$\mathrm{SG} \subsetneq \mathrm{CM}_{\mathrm{deph}} \subsetneq \mathrm{CP\text{-}Div} \subsetneq \mathrm{P\text{-}Div} \subsetneq \mathrm{Spohn} \subsetneq \mathrm{CPTP}$}{SG < CM\_deph < CP-Div < P-Div < Spohn < CPTP}}

For continuous one-parameter families of qubit CPTP channels $\{\Phi_t\}_{t\ge 0}$ with $\Phi_0 = \mathrm{id}$ and at least one invariant state $\Phi_t(\rho_*) = \rho_*$:
\begin{itemize}[leftmargin=1.6em,itemsep=2pt]
\item $\mathrm{SG} \subseteq \mathrm{CM}_{\mathrm{deph}}$ (on the pure-dephasing/isotropic tier $G(t) = \ee^{-\gamma t}$): $(-1)^n \frac{\dd^n}{\dd t^n}\ee^{-\gamma t} = \gamma^n \ee^{-\gamma t} \ge 0$ for all $n \ge 0, t > 0$.
\item $\mathrm{CM}_{\mathrm{deph}} \subseteq \mathrm{CP\text{-}Div}$: for a single-axis or isotropic channel with completely monotonic coherence factor $G(t) \in (0,1]$, $(-1)^1\dot{G}(t) \ge 0$ guarantees canonical GKSL rate $\gamma(t) = -\dot{G}(t)/G(t) \ge 0$. \emph{Multi-axis caveat:} componentwise complete monotonicity of Bloch multipliers $(\Lambda_x, \Lambda_y, \Lambda_z)$ does \emph{not} imply CP-divisibility for general Pauli channels, as demonstrated directly by Level~4a (Eternal Pauli, $\Lambda_x=\Lambda_y=\frac{1}{2}(1+\ee^{-2t}), \Lambda_z=\ee^{-2t}$) and Level~4b (Subordinated Pauli at $\mu=0.95$, $\Lambda_k(t) = E_{0.95}(-\lambda_k t^{0.95})$, CM by Pollard's theorem and Bernstein composition).
\item $\mathrm{CP\text{-}Div} \subseteq \mathrm{P\text{-}Div}$: every completely positive trace-preserving intermediate propagator $\Lambda_{t,s}$ is positive and trace-preserving.
\item $\mathrm{P\text{-}Div} \subseteq \mathrm{Spohn}$: here $\mathrm{P\text{-}Div}$ is defined strictly as positivity of every trace-preserving intermediate propagator $\Lambda_{t,s}$ (equivalent for invertible maps to biased Helstrom monotonicity~\cite{Wissmann2015PRA,Bae2016PRL}; coinciding with unbiased $\NBLP = 0$ on unital qubit channels and strictly stronger than $\{\NBLP = 0\}$ on non-unital channels). Whenever $\Lambda_{t,s}$ is positive, trace-preserving, and fixes $\rho_*$, the M\"uller-Hermes--Reeb data-processing inequality~\cite{MullerHermes2017AHP} yields $D(\Phi_t(\rho)\parallel\rho_*) = D(\Lambda_{t,s}(\Phi_s(\rho))\parallel\Lambda_{t,s}(\rho_*)) \le D(\Phi_s(\rho)\parallel\rho_*)$ for every initial $\rho$ and all $t \ge s \ge 0$.
\item $\mathrm{Spohn} \subseteq \mathrm{CPTP}$ holds by definition of the ambient family.
\end{itemize}

\begin{equation}
\label{eq:S5_level5}
T(t) = \begin{cases}
\big(1 - \frac{19}{20}t\big) I_3, & 0 \le t \le 1, \\[4pt]
\frac{1}{20}\,\ee^{(t-1)K}, & t \ge 1,
\end{cases}
\qquad
K = \begin{pmatrix} -a & 0 & c \\ 0 & -a & 0 \\ 0 & 0 & -b \end{pmatrix},
\end{equation}
with $a = \ln 2$, $b = \ln\frac{5}{2}$, and $c = \frac{9}{5}$. We verify all four properties:
\begin{enumerate}[label=(\roman*),leftmargin=1.8em,itemsep=2pt]

\label{eq:S5_expK}

\ee^{-a\tau} & 0 & \frac{c}{b-a}(\ee^{-a\tau} - \ee^{-b\tau}) \\
0 & \ee^{-a\tau} & 0 \\
0 & 0 & \ee^{-b\tau}

Maximizing $\|\ee^{\tau K}\|_2$ over $\tau \ge 0$ yields $\sup_{\tau\ge 0}\|\ee^{\tau K}\|_2 = 1.038658$ (attained at $\tau = 0.58$). Because every initial Bloch vector satisfies $\|\mathbf{r}_0\|_2 \le 1$, we have $\|\mathbf{r}_0 - \mathbf{r}_*\|_2 \le 1 + \|\mathbf{r}_*\|_2 = 1.8$ (attained at the south pole $\mathbf{r}_0 = (0,0,-1)^\top$). \|\mathbf{r}(t) - \mathbf{r}_*\|_2 \le \frac{1}{20}\cdot 1.8\,\sup_{\tau\ge 0}\|\ee^{\tau K}\|_2 = 0.093479 < 0.095
\end{equation}
(direct optimization over all $\|\mathbf{r}_0\|_2 \le 1$ and $\tau \ge 0$ yields exact maximum distance $0.089991 \le 0.090$ and Bloch norm range $\|\mathbf{r}(t)\|_2 \in [0.7100, 0.8100]$).
\item \textbf{Analytic Global Complete Positivity (CPTP) for All $t \ge 0$:} On $t \in [0, 1]$, $\Phi_t(\rho_0) = (1 - \frac{19}{20}t)\rho_0 + \frac{19}{20}t\,\rho_*$ is a convex combination of CPTP maps with $\min_{t\in[10^{-3},1]}\lambda_{\min}(J(\Phi_t)) = 4.750\times 10^{-5} > 0$. On $t \ge 1$ ($\tau = t - 1 \ge 0$), write $T(t) = \mathrm{diag}(a(\tau), a(\tau), c(\tau)) + b(\tau)\,E_{13}$ with $a(\tau) = \frac{1}{20}2^{-\tau} \in (0, \frac{1}{20}]$, $c(\tau) = \frac{1}{20}(\frac{5}{2})^{-\tau} \in (0, \frac{1}{20}]$, and $b(\tau) = \frac{1}{20}\frac{9/5}{\ln(5/4)}(2^{-\tau} - (\frac{5}{2})^{-\tau}) \in [0, 0.041277] \subset [0, \frac{1}{20}]$ (attaining its unique maximum at $\tau = \frac{\ln(\ln(5/2)/\ln 2)}{\ln(5/4)} = 1.2464$). With exact fixed-point shift $m_x = -\frac{4}{5}b$ and $m_z = \frac{4}{5}(1-c)$, the real symmetric $4\times 4$ Choi matrix $J(\Phi_t)$ has diagonal entries $s_{00} = (1+m_z+c)/4 \ge 0.45$, $s_{11} = (1-m_z-c)/4 \ge 0.0475$, $s_{22} = (1+m_z-c)/4 \ge 0.4275$, $s_{33} = (1-m_z+c)/4 \ge 0.05$, and only three distinct off-diagonal entries $s_{01} = (m_x+b)/4 = b/20 \in [0, 0.0025]$, $s_{03} = a/2 \in [0, 0.025]$, and $s_{23} = (m_x-b)/4 = -9b/20 \in [-0.0225, 0]$. \[
D\big(\Phi_t(\rho_0)\parallel\rho_*\big)
\le \lambda\,D\big(\Phi_s(\rho_0)\parallel\rho_*\big) + (1-\lambda)\,D(\rho_*\parallel\rho_*)
= \lambda\,D\big(\Phi_s(\rho_0)\parallel\rho_*\big)
\le D\big(\Phi_s(\rho_0)\parallel\rho_*\big).
\]
On $t \ge 1$, writing $\rho = \frac{1}{2}(I + \mathbf{r}\cdot\boldsymbol{\sigma})$ with $\ln\rho = \frac{1}{2}\ln\!\big(\frac{1-|\mathbf{r}|^2}{4}\big)I + \frac{\operatorname{artanh}|\mathbf{r}|}{|\mathbf{r}|}(\mathbf{r}\cdot\boldsymbol{\sigma})$ and using $\Tr(\dot{\rho}) = 0$ and $\Tr(\sigma_i\sigma_j) = 2\delta_{ij}$, the exact relative-entropy production rate reads
\begin{equation}
\label{eq:S5_dD_dt}
\frac{\dd}{\dd t} D\big(\rho(t)\parallel\rho_*\big)
= \Tr\Big[\dot{\rho}(t)\big(\ln\rho(t) - \ln\rho_*\big)\Big]
= \big(\mathbf{r}(t) - \mathbf{r}_*\big)^\top K^\top
\left(\frac{\operatorname{artanh}|\mathbf{r}(t)|}{|\mathbf{r}(t)|}\mathbf{r}(t)
- \frac{\operatorname{artanh} z_*}{z_*}\mathbf{r}_*\right).
\end{equation}
Linearizing Eq.~\eqref{eq:S5_dD_dt} around $\mathbf{r}_*$ via $\delta := \mathbf{r} - \mathbf{r}_*$ gives the Bogoliubov--Kubo--Mori (BKM) Fisher quadratic form $\frac{\dd D}{\dd t} = \frac{1}{2}\delta^\top(K^\top H_* + H_* K)\delta + O(\|\delta\|_2^3)$, where $H_* = \mathrm{diag}(g_\perp, g_\perp, g_\parallel)$ with $g_\perp = \frac{\operatorname{artanh}(4/5)}{4/5} = \frac{5}{4}\ln 3 \approx 1.373265$ and $g_\parallel = \frac{1}{1 - (4/5)^2} = \frac{25}{9} \approx 2.777778$. The symmetrized BKM flow matrix $M_{\mathrm{BKM}} := K^\top H_* + H_* K$ has strictly negative eigenvalues $\{-6.438050, -1.903750, -0.556204\}$ (Sylvester negative-definiteness interval-certified in Lean~4 Theorem~\leanid{sep\_pdiv\_spohn\_bkm\_vs\_blp}). Nonlinear global optimization of Eq.~\eqref{eq:S5_dD_dt} across the entire forward-invariant ball Eq.~\eqref{eq:S5_inv_ball} confirms strict negativity throughout:
\begin{itemize}[leftmargin=1.4em,itemsep=1pt]
\item On the initial pure Bloch sphere $|\mathbf{r}_0| = 1$ at $t = 1^+$: $\max_{|\mathbf{r}_0|=1}\frac{\dd D}{\dd t}(1^+) = -7.2260\times 10^{-5} < 0$.

\end{itemize}
Hence $\frac{\dd}{\dd t} D(\Phi_t(\rho_0)\parallel\rho_*) \le 0$ for every initial qubit state $\rho_0 \in B_1$ and every $t \ge 0$, proving $\Phi_t \in \mathrm{Spohn} \setminus \mathrm{P\text{-}Div}$.
\end{enumerate}

\section{Unitary \texorpdfstring{$N = 1000$}{N = 1000} Friedrichs--Lee Discretization and Continuum Volterra \texorpdfstring{$E_{2-\mu}$}{E\_\{2-mu\}} Derivation}
\label{sec:S6}

For the single-excitation Friedrichs--Lee Hamiltonian $H = \omega_0|e\rangle\langle e| + \sum_{k=1}^N \omega_k b_k^\dagger b_k + \sum_{k=1}^N g_k(|e\rangle\langle g|b_k + \mathrm{H.c.})$, writing $|\psi(t)\rangle = c(t)|e,\mathrm{vac}\rangle + \sum_{k=1}^N b_k(t)|g, 1_k\rangle$ with $c(0)=1, b_k(0)=0$ yields $\dot{c} = -\ii\omega_0 c - \ii\sum_k g_k b_k$ and $\dot{b}_k = -\ii\omega_k b_k - \ii g_k c$. Integrating $b_k(t) = -\ii g_k\int_0^t \ee^{-\ii\omega_k(t-\tau)} c(\tau)\,\dd\tau$ and passing to the continuum $\sum_k g_k^2 \delta(\omega-\omega_k) \to J(\omega)$ gives the exact Volterra equation
\begin{equation}
\label{eq:S6_volterra}
\dot{c}(t) + \ii\omega_0 c(t) = -\int_0^t \mathcal{K}(t-\tau)\,c(\tau)\,\dd\tau, \qquad
\mathcal{K}(t) := \int J(\omega)\,\ee^{-\ii\omega t}\,\dd\omega, \qquad
\tilde{c}(s) = \frac{1}{s + \ii\omega_0 + \tilde{\mathcal{K}}(s)}.
\end{equation}
Tracing over the bath yields a generalized amplitude-damping channel on the emitter ($\rho_{ee}(t) = |c(t)|^2\rho_{ee}(0)$, $\rho_{eg}(t) = c(t)\rho_{eg}(0)$), for which any initial state pair with population difference $a \in [-1,1]$ and coherence difference $b \in \mathbb{C}$ ($a^2 + |b|^2 \le 1$) has trace distance $D_{\mathrm{tr}}(t) = \sqrt{a^2|c(t)|^4 + |b|^2|c(t)|^2}$. When $c(t)$ passes through zero (Lorentzian and symmetric broadband baths), antipodal equatorial states ($a=0, |b|=1$, $D_{\mathrm{tr}}=|c(t)|$) are optimal; conversely, when a one-sided bound state traps $|c(t)| > 1/2$, $\frac{\dd}{\dd t}(|c|^2) = 2|c|\frac{\dd|c|}{\dd t} > \frac{\dd|c|}{\dd t}$, so the excited/ground pair $|e\rangle, |g\rangle$ ($a=1, b=0$, $D_{\mathrm{tr}} = |c(t)|^2$) strictly maximizes $\NBLP^{\mathrm{AD}}$ (e.g., $0.2998$ vs.\ $0.2757$ at $(\mu,\omega_0)=(0.50,0)$ and $0.2234$ vs.\ $0.1794$ at $(0.75,0)$, fine grid $\Delta t = 0.002$ on $t \in [0,20]$). For the three bath spectra studied in Section~V of the main text:
\begin{enumerate}[label=(\roman*),leftmargin=1.8em,itemsep=2pt]
\item \textbf{Resonant Lorentzian ($J(\omega) = \frac{2g}{\pi[(\omega-\omega_0)^2 + 4g^2]}$, $\omega_0 = 25$):} On $\mathbb{R}$, $\mathcal{K}_{\mathrm{rot}}(t) = \ee^{\ii\omega_0 t}\mathcal{K}(t) = \ee^{-2gt}$, giving $\tilde{\mathcal{K}}_{\mathrm{rot}}(s) = (s+2g)^{-1}$ and $\tilde{c}_{\mathrm{rot}}(s) = [s + (s+2g)^{-1}]^{-1} = \frac{s+2g}{s^2+2gs+1} = \tilde{G}_{1,g}(s)$.
\item \textbf{One-sided power-law ($J(\omega) = \frac{\eta}{\Gamma(\mu)}\omega^{\mu-1}\ee^{-\omega/\omega_c}$ on $\omega > 0$):} Direct Gamma-function integration gives $\mathcal{K}(t) = \eta(\omega_c^{-1} + \ii t)^{-\mu}$. Because $\int_0^\infty J(\omega)/\omega\,\dd\omega = \infty$ for $\mu \in (0,1)$, the real-frequency self-energy $\Sigma(E) = \int_0^\infty \frac{J(\omega)}{E-\omega}\,\dd\omega$ diverges to $-\infty$ as $E \to 0^-$, guaranteeing a discrete real-frequency bound state $E_0 < 0$ solving $E_0 - \omega_0 = \Sigma(E_0)$ for every $\eta > 0$ and $\omega_0 \in \mathbb{R}$.
\item \textbf{Symmetric broadband power-law ($J_{\mathrm{sym}}(\omega) = \frac{\eta}{\Gamma(\mu)}|\omega-\omega_0|^{\mu-1}\ee^{-|\omega-\omega_0|/\omega_c}$):} Shifting $\Omega = \omega - \omega_0$ gives $\mathcal{K}_{\mathrm{rot}}(t) = 2\eta\operatorname{Re}[(\omega_c^{-1} + \ii t)^{-\mu}] \xrightarrow{\omega_c\to\infty} 2\eta\cos(\frac{\pi\mu}{2}) t^{-\mu}$. Its Laplace transform is $\tilde{\mathcal{K}}_{\mathrm{rot}}(s) = c_{\mathrm{sym}} s^{\mu-1}$ with $c_{\mathrm{sym}} := 2\eta\cos(\frac{\pi\mu}{2})\Gamma(1-\mu)$, yielding
\begin{equation}
\label{eq:S6_E2minusmu}
\tilde{c}_{\mathrm{rot}}(s) = \frac{1}{s + c_{\mathrm{sym}} s^{\mu-1}} = \frac{s^{(2-\mu)-1}}{s^{2-\mu} + c_{\mathrm{sym}}}
\quad\Longleftrightarrow\quad
c_{\mathrm{rot}}(t) = E_{2-\mu}\!\left(-c_{\mathrm{sym}}\,t^{2-\mu}\right),
\end{equation}
which has Mittag-Leffler index $2-\mu \in (1,2)$ (oscillatory with coherent revivals $\NBLP > 0$), contrasting sharply with inverse-stable operational-time subordination $E_\mu(-c t^\mu)$ (index $\mu \in (0,1)$, completely monotonic with $\NBLP = 0$).
\end{enumerate}
